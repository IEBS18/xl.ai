    • Python Implementation


    import scipy.stats as stats
    

    

    ANOVA test

    
anova_result = stats.f_oneway(df[df['category'] == 'A']['sales'],
    
                               df[df['category'] == 'B']['sales'],
    
                               df[df['category'] == 'C']['sales'])
    
print(f'ANOVA F-statistic: {anova_result.statistic}, p-value: {anova_result.pvalue}')
    

    

    Confidence intervals

    
mean_sales = df.groupby('category')['sales'].mean()
    
conf_intervals = df.groupby('category')['sales'].agg(lambda x: stats.t.interval(0.95, len(x)-1, loc=np.mean(x), scale=stats.sem(x)))
    
print("Confidence Intervals:\n", conf_intervals)

    from statsmodels.tsa.arima.model import ARIMA
    
from xgboost import XGBRegressor
    
from sklearn.metrics import mean_squared_error
    

    

    Split data into training and test sets

    
train_size = int(len(df) * 0.8)
    
train, test = df[:train_size], df[train_size:]
    

    

    ARIMA model

    
arima_model = ARIMA(train['sales'], order=(1, 1, 1))
    
arima_fit = arima_model.fit()
    
arima_forecast = arima_fit.forecast(steps=len(test))
    

    

    XGBoost model

    
xgb_model = XGBRegressor()
    
xgb_model.fit(train[['sales_lag1', 'sales_lag2']], train['sales'])
    
xgb_forecast = xgb_model.predict(test[['sales_lag1', 'sales_lag2']])
    

    

    Evaluate models

    
arima_rmse = np.sqrt(mean_squared_error(test['sales'], arima_forecast))
    
xgb_rmse = np.sqrt(mean_squared_error(test['sales'], xgb_forecast))
    
print(f'ARIMA RMSE: {arima_rmse}, XGBoost RMSE: {xgb_rmse}')
