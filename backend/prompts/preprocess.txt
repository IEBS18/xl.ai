import pandas as pd
import numpy as np
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import MinMaxScaler
from statsmodels.tsa.stattools import adfuller

# assume `df` is your cleaned DataFrame with columns ['Date', 'Value'] + metadata

# 0. RENAME for consistency
df = df.rename(columns={'Date': 'ds', 'Value': 'y'})

# 1. VALIDATION & REGULARIZATION
df['ds'] = pd.to_datetime(df['ds'])
df = df.sort_values('ds').reset_index(drop=True)
df['y'] = pd.to_numeric(df['y'], errors='raise')

# infer freq and reindex to a complete timeline
freq = df['ds'].diff().mode().iat[0]
full_idx = pd.date_range(df['ds'].min(), df['ds'].max(), freq=freq)
df = (
    df.set_index('ds')
      .reindex(full_idx)
      .rename_axis('ds')
      .reset_index()
)
df['y'] = SimpleImputer(strategy='median').fit_transform(df[['y']])

# 2. FEATURE ENGINEERING

# 2a. Classical TS (ARIMA, SARIMA, Prophet)
# – stationarity
adf_p = adfuller(df['y'])[1]
if adf_p > 0.05:
    df['y_diff'] = df['y'].diff()
    target = 'y_diff'
else:
    target = 'y'

# – calendar features
df['year']        = df['ds'].dt.year
df['month']       = df['ds'].dt.month
df['quarter']     = df['ds'].dt.quarter
df['day_of_week'] = df['ds'].dt.dayofweek
df['day_of_month']= df['ds'].dt.day

# – lags & rolling stats
df['lag_1']          = df[target].shift(1)
df['rolling_mean_7'] = df[target].rolling(window=7).mean()
df['rolling_std_7']  = df[target].rolling(window=7).std()

# 2b. Machine-learning (LinReg, XGB, RF)
# cyclical encoding
df['month_sin'] = np.sin(2 * np.pi * df['month'] / 12)
df['month_cos'] = np.cos(2 * np.pi * df['month'] / 12)
df['dow_sin']   = np.sin(2 * np.pi * df['day_of_week'] / 7)
df['dow_cos']   = np.cos(2 * np.pi * df['day_of_week'] / 7)

# 2c. Deep-learning (LSTM)
scaler = MinMaxScaler()
df['y_scaled'] = scaler.fit_transform(df[['y']])

def create_sequences(array, timesteps=5):
    X, y = [], []
    for i in range(len(array) - timesteps):
        X.append(array[i:i+timesteps])
        y.append(array[i+timesteps])
    return np.array(X), np.array(y)

X_lstm, y_lstm = create_sequences(df[['y_scaled']].values, timesteps=5)
# reshape to [samples, timesteps, features]
X_lstm = X_lstm.reshape(X_lstm.shape[0], X_lstm.shape[1], 1)

# 3. FINAL VALIDATION & SUMMARY
df_final = df.dropna().reset_index(drop=True)

print("Final DataFrame shape:", df_final.shape)
print("Features:", df_final.columns.tolist())
print("LSTM X shape:", X_lstm.shape, "LSTM y shape:", y_lstm.shape)

# df_final and (X_lstm, y_lstm) are ready for modeling
